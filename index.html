// ═══ UTILITY DATE ═══

function isFest(d){const mm=String(d.getMonth()+1).padStart(2,'0'),gg=String(d.getDate()).padStart(2,'0');return d.getDay()===0||['01-01','06-01','04-25','05-01','06-02','08-15','11-01','12-08','12-25','12-26'].includes(`${mm}-${gg}`);}

function isSab(d){return d.getDay()===6;}

function nextLav(d,es){const n=new Date(d);while(isFest(n)||(es&&isSab(n)))n.setDate(n.getDate()+1);return n;}

function addCal(b,n,es){const d=new Date(b);d.setDate(d.getDate()+n);return nextLav(d,es);}

function fmt(d){return `${String(d.getDate()).padStart(2,'0')}/${String(d.getMonth()+1).padStart(2,'0')}/${d.getFullYear()}`;}

function sospFer(d){

  if(d.getMonth()!==7)return{d,nota:null};

  return{d:nextLav(new Date(d.getFullYear(),8,1),false),nota:'Termine in agosto: sospeso (L. 742/1969) e prorogato al 1° settembre.'};

}

function giLiberi(da,a,es){

  let c=0;const cur=new Date(da);cur.setDate(cur.getDate()+1);

  const fine=new Date(a);fine.setDate(fine.getDate()-1);

  while(cur<=fine){if(!isFest(cur)&&(!es||!isSab(cur)))c++;cur.setDate(cur.getDate()+1);}

  return c;

}

function giCal(a,b){return Math.round((new Date(b)-new Date(a))/(864e5));}

function buildVerd(pD,udS,tcMin,es,tipo){

  if(!udS)return'';

  const ud=new Date(udS+'T12:00:00'),today=new Date();today.setHours(12,0,0,0);

  const lib=giLiberi(pD,ud,es),alUd=giCal(today,ud),alN=giCal(today,pD);

  const ok=lib>=tcMin&&ud>pD;

  const acc=tipo==='c'?'#1a6fa8':'var(--rust)';

  let cls,ico,tit,sub;

  if(ud<=pD){cls='ko';ico='🚫';tit='Udienza precedente alla notifica';sub='Impossibile: l\'udienza non può precedere il perfezionamento della notifica.';}

  else if(!ok){cls='ko';ico='❌';tit=`Termini INSUFFICIENTI – mancano ${tcMin-lib} gg liberi`;sub=`Intercorrono ${lib} gg liberi su ${tcMin} richiesti. Rinviare l'udienza o accelerare la notifica.`;}

  else if(lib<tcMin+3){cls='warn';ico='⚠️';tit='Termini rispettati – margine stretto';sub=`${lib} gg liberi (min. ${tcMin}). Margine di soli ${lib-tcMin} gg: verificare la modalità di notifica.`;}

  else{cls='ok';ico='✅';tit='Termini regolari';sub=`${lib} gg liberi intercorrenti, ${lib-tcMin} in più del minimo di ${tcMin} richiesti.`;}

  const cUd=alUd<0?'r':(alUd<5?'a':'g'),cN=alN<0?'r':(alN<2?'a':'g');

  return`<div class="verdict ${cls}"><div class="verdict-ico">${ico}</div><div class="verdict-body"><div class="verdict-title">${tit}</div><div class="verdict-sub">${sub}</div></div></div>

  <div class="days-row">

    <div class="dc"><div class="n ${lib>=tcMin?'g':'r'}">${lib}</div><div class="l">Gg liberi<br>intercorrenti</div></div>

    <div class="dc"><div class="n" style="color:${acc}">${tcMin}</div><div class="l">Minimo<br>richiesto</div></div>

    <div class="dc"><div class="n ${cUd}">${alUd>=0?alUd:0}</div><div class="l">Gg all'<br>udienza</div></div>

    <div class="dc"><div class="n ${cN}">${alN>=0?alN:'scad.'}</div><div class="l">Gg alla<br>notifica</div></div>

  </div>`;

}
